1. Back up the database
docker exec sitelink_postgres pg_dump -U sitelink -d sitelink_db -Fc \
  > sitelink_before_backfill_$(date +%Y%m%d-%H%M%S).dump

2. Run a SQL statement
docker exec -i sitelink_postgres psql -U sitelink -d sitelink_db \
  -v ON_ERROR_STOP=1 -1 -f - \
  < backend/scripts/one_off_backfill_site_cells.sql
